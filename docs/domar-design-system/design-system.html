04 Install

Two files ship alongside this page. Drop them into your React / Next.js / Tailwind v4 app.
// 1. Drop domar-ds.css into your project (e.g. src/styles/).
// 2. Import once at the root of your app:

// src/main.tsx
import './styles/domar-ds.css';
import ReactDOM from 'react-dom/client';
import App from './App';

ReactDOM.createRoot(document.getElementById('root')!)
  .render(<App />);

// 3. Use the classes directly in JSX:

function KpiTile() {
  return (
    <div className="dom-kpi">
      <div className="dom-kpi-label">Today's jobs</div>
      <div className="dom-kpi-value">8</div>
      <div className="dom-kpi-delta up">
        <span className="dot" /> +2 vs yesterday
      </div>
    </div>
  );
}
// Next.js (App Router): app/layout.tsx

import './globals.css';             // your existing global styles
import '../styles/domar-ds.css';   // 👈 the design system

export default function RootLayout({
  children
}: { children: React.ReactNode }) {
  return (
    <html lang="en">
      <body>{children}</body>
    </html>
  );
}

// Anywhere in your app — Server or Client Component:

export function AttentionBanner({ items }: { items: Item[] }) {
  return (
    <section className="dom-attention">
      <div className="dom-attention-head">
        <div className="dom-attention-title">
          <span className="dom-pulse" />
          Needs your attention
        </div>
      </div>
      {items.map(i => (
        <div key={i.id} className="dom-att-card"> … </div>
      ))}
    </section>
  );
}
/* Tailwind v4 reads its theme from CSS, not JS.   */
/* In your app/globals.css (or src/index.css):    */

@import "tailwindcss";
@import "./domar-tailwind.css";  /* 👈 maps tokens to Tailwind */

/* Now use them as Tailwind utilities: */

function NewJobButton() {
  return (
    <button className="bg-accent text-white px-3.5 py-2
                       rounded-md font-medium shadow-sm
                       hover:bg-accent/90 transition">
      New job
    </button>
  );
}

function InProgressPill() {
  return (
    <span className="inline-flex items-center gap-1.5
                     bg-st-progress-soft text-st-progress
                     border border-st-progress/30
                     rounded-full px-2 py-0.5 text-[11px]
                     font-semibold uppercase tracking-wide">
      <span className="size-1.5 rounded-full bg-st-progress
                       animate-pulse" />
      In progress
    </span>
  );
}

/* Available utilities after import:
   bg-bg            bg-surface       bg-surface-2
   text-fg          text-muted       text-subtle
   border-border    border-border-strong
   bg-accent        text-accent-fg   ring-accent-line
   bg-st-progress   text-st-declined …all status colors
   font-mono        text-md / xl / 2xl / 3xl
   rounded-sm / md / lg / pill
   shadow-sm / md / lg
*/
<!-- 1. Drop domar-ds.css next to your HTML file. -->
